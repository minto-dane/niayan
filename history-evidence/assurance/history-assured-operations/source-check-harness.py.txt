def run(name,args,cwd,success=True):
    q=subprocess.run(args,cwd=cwd,text=True,stdout=subprocess.PIPE,stderr=subprocess.STDOUT,timeout=25)
    (E/(name+'.log')).write_text('$ '+' '.join(args)+'\n'+q.stdout+'\nexit='+str(q.returncode)+'\n')
    if success:
        assert q.returncode==0,(name,q.stdout)
        rows.append(dict(name=name,result='PASS',kind='source-or-build-entry'))
    else:
        blocked.append(dict(name=name,exit=q.returncode,status='BLOCKED_TOOLCHAIN' if q.returncode else 'UNEXPECTED_AVAILABLE',log=name+'.log'))
    return q
