def sort_services(n,req,after,selected):
    result=[];pending=set(selected)
    while pending:
        free=sorted(i for i in pending if not ((req[i]|after[i])&pending))
        require(free)
        for i in free:pending.remove(i);result.append(i)
    return result
def closure(selected,edges):
    s=set(selected)
    for _ in range(len(edges)):
        s|=set().union(*(edges[i] for i in list(s))) if s else set()
    return s

def net(s,event,now,seq=None,healthy=True,armed=True,correlated=True):
    t=copy.deepcopy(s);require(now>=s['last'] and s['revision']<U63);seq=seq or s['seq']+1;require(seq>s['seq']);effect=None
    require(s['phase'] not in ('committed','restored'))

    elif event=='checkpoint':require(s['phase']=='checkpoint_pending' and correlated and armed and now<890);t['phase']='checkpoint_ready'
    elif event=='apply':require(s['phase']=='checkpoint_ready' and armed and s['expiry']-now>10 and s['expiry']-now-10>=20);t['phase']='apply_pending';effect='apply'
    elif event=='applied':require(s['phase']=='apply_pending' and correlated and now<s['expiry']);t['phase']='verifying';t['samples']=0
    elif event=='path':
        require(s['phase'] in ('verifying','commit_pending') and now>s['last_sample'] and now>=s['effect_at'] and now<s['expiry'])
        if not healthy or not armed:
            t.update(samples=0,last_sample=now,since=now)
            if s['phase']=='commit_pending':t['phase']='reconcile'
        else:
            require(s['phase']=='verifying')
            if s['samples']==0 or now-s['last_sample']>15:t.update(samples=1,since=now)
            else:t['samples']=min(255,s['samples']+1)
            t['last_sample']=now
    elif event=='commit':
        require(s['phase']=='verifying' and healthy and armed and s['samples']>=3 and now-s['last_sample']<=15 and s['last_sample']-s['since']>=20 and s['expiry']-now>10)
        t['phase']='commit_pending';effect='destroy'
    elif event=='committed':require(s['phase']=='commit_pending' and healthy and correlated);t['phase']='committed'
    elif event in ('rollback','deadline'):

        if s['phase'] in ('commit_pending','checkpoint_pending','reconcile'):t['phase']='reconcile';t['samples']=0
        else:require(s['phase'] in ('checkpoint_ready','apply_pending','verifying'));t['phase']='rollback_pending';effect='rollback';t['samples']=0

    else:raise AssertionError('unknown event')
